export function catalogCategories(products = []) {
  const categories = new Map()
  for (const product of products) {
    const id = categoryIdOf(product)
    const label = categoryNameOf(product)
    if (!id || !label) continue
    const current = categories.get(id) || { id, label, count: 0, soldCount: 0 }
    current.count += 1
    current.soldCount += Number(product?.sold_count) || 0
    categories.set(id, current)
  }
  return [...categories.values()]
    .filter(category => category.count > 0 && normalizedText(category.label) !== 'khac')
    .sort((left, right) => right.soldCount - left.soldCount || left.label.localeCompare(right.label, 'vi'))
}

function replaceHomeCards(container, documentRef, categories) {
  if (!container) return
  const prototype = container.firstElementChild
  const fragment = documentRef.createDocumentFragment()
  for (const category of categories.slice(0, 6)) {
    const card = documentRef.createElement('a')
    card.className = prototype?.className || 'desk-cat-card'
    card.href = categoryHref(category.id)
    if (prototype) {
      for (const child of [...prototype.cloneNode(true).childNodes]) card.appendChild(child)
    }
    const label = card.querySelector('span:last-child') || card.appendChild(documentRef.createElement('span'))
    label.textContent = category.label
    fragment.appendChild(card)
  }
  container.replaceChildren(fragment)
}
