Relevant current code excerpt: gallery selection
```js
function renderCarouselGallery(images, name) {
  const carousel = document.querySelector('.carousel-inner')
  const slides = [...document.querySelectorAll('.carousel-slide')]
  const thumbs = [...document.querySelectorAll('.thumb-item')]
  let current = 0
  const select = index => {
    current = (index + images.length) % images.length
    carousel.style.transform = `translateX(-${current * 100}%)`
    thumbs.forEach((thumb, thumbIndex) => thumb.classList.toggle('active', thumbIndex === current))
  }
  thumbs.forEach((thumb, index) => {
    makeInteractive(thumb, `Xem ảnh ${index + 1} của ${name}`, () => select(index))
  })
  select(0)
}
```
